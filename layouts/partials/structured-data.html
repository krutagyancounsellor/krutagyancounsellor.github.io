{{- /* JSON-LD structured data. - `Person` / `LocalBusiness` graph on every page, carrying the email, phone, WhatsApp URL and `sameAs` social profiles. `sameAs` is how search engines tie the website to its social profiles. - `Offer` schema on the homepage, describing the session fee, duration and the complimentary initial consultation. - `Article` schema on blog posts. - `BreadcrumbList` schema on nested pages. */ -}} {{- $author := .Site.Params.authorFullName -}} {{- $socialURLs := slice .Site.Params.instagram.url -}} {{- range .Site.Params.socials -}} {{- $socialURLs = $socialURLs | append .url -}} {{- end -}} {{- /* ---------- Article schema for blog posts ---------- */ -}} {{- if and (eq .Kind "page") (eq .Section "blog") -}} {{- $desc := .Params.description | default (.Summary | plainify | htmlUnescape | truncate 155) -}} {{- $imgURL := "" -}} {{- with .Params.image -}} {{- with resources.Get . -}} {{- $imgURL = printf "%s%s" $.Site.BaseURL (strings.TrimPrefix "/" .RelPermalink) -}} {{- end -}} {{- end -}} {{- $article := dict "@context" "https://schema.org" "@type" "BlogPosting" "@id" .Permalink "headline" .Title "description" $desc "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "author" (dict "@type" "Person" "name" $author "url" .Site.BaseURL) "publisher" (dict "@type" "Organization" "name" .Site.Title "url" .Site.BaseURL) "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "inLanguage" "en-IN" "keywords" (delimit .Params.tags ", ") "articleSection" "Mental Health & Counselling" -}} {{- if $imgURL -}} {{- $article = merge $article (dict "image" $imgURL) -}} {{- end -}} {{- end -}} {{- /* ---------- Offer schema (homepage) ---------- Describes what the site actually offers: the session duration, the sliding-scale fee, and — while the complimentary offer is live — the introductory consultation. Emitted on the homepage only. */ -}} {{- if .IsHome -}} {{- $session := .Site.Params.session -}} {{- $offer := .Site.Params.offer -}} {{- $on := false -}} {{- with $offer -}}{{- $on = .enabled | default false -}}{{- end -}} {{- $offerNode := dict "@context" "https://schema.org" "@type" "Offer" "@id" (printf "%s#offer" .Site.BaseURL) "name" (cond $on (i18n "cta.label") (i18n "cta.label_off")) "description" (cond $on (i18n "offer.disclaimer") (i18n "suitability.booking_body_off")) "priceCurrency" "INR" "price" "1200" "priceSpecification" (dict "@type" "PriceSpecification" "minPrice" "1200" "maxPrice" "2000" "priceCurrency" "INR" "valueAddedTaxIncluded" false ) "availability" "https://schema.org/InStock" "areaServed" "IN" "url" .Site.Params.bookingURL -}} {{- if $on -}} {{- $offerNode = merge $offerNode (dict "price" "0" "description" (i18n "offer.disclaimer") ) -}} {{- end -}} {{- $offerList := dict "@context" "https://schema.org" "@type" "OfferCatalog" "name" (i18n "section.session_heading") "itemListElement" (slice (dict "@type" "Offer" "name" (i18n "session.duration_label") "description" $session.duration "itemOffered" (dict "@type" "Service" "name" (i18n "session.duration_label") "serviceType" "Counselling" "provider" (dict "@id" (printf "%s#business" .Site.BaseURL)) ) )) -}} {{- end -}} {{- /* ---------- BreadcrumbList schema ---------- */ -}} {{- if not .IsHome -}} {{- $crumbs := slice (dict "@type" "ListItem" "position" 1 "name" "Home" "item" .Site.BaseURL ) -}} {{- $pos := 2 -}} {{- if .Section -}} {{- $parent := .Parent -}} {{- if and $parent (ne $parent .Site.Home) -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" $pos "name" $parent.Title "item" $parent.Permalink) -}} {{- $pos = add $pos 1 -}} {{- end -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" $pos "name" .Title "item" .Permalink) -}} {{- end -}}
