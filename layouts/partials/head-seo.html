{{- /* Meta description, Open Graph, Twitter Card and canonical URL tags. Every value falls back to a site-wide default from `hugo.yaml` params so no page is ever published without a description. */ -}} {{- $page := . -}} {{- $desc := "" -}} {{- if $page.Params.description -}} {{- $desc = $page.Params.description -}} {{- else if $page.IsHome -}} {{- $desc = $page.Site.Params.description -}} {{- else if $page.Summary -}} {{- $desc = $page.Summary | plainify | htmlUnescape | truncate 155 -}} {{- else -}} {{- $desc = $page.Site.Params.description -}} {{- end -}} {{- /* Page titles: the homepage uses the bare brand name, other pages use "Title | Brand". Keep the whole thing under ~60 chars where possible. */ -}} {{- $title := "" -}} {{- if $page.IsHome -}} {{- $tagline := $page.Site.Params.homeTitle | default "" -}} {{- $title = printf "%s%s" $page.Site.Title (cond (ne $tagline "") (printf " | %s" $tagline) "") -}} {{- else -}} {{- $title = printf "%s | %s" $page.Title $page.Site.Title -}} {{- end -}} {{- /* Social share image: the page's own image if it has one, otherwise the site logo. Must be an absolute URL for crawlers to fetch it. */ -}} {{- $imgResource := "" -}} {{- with $page.Params.image -}} {{- $imgResource = resources.Get . -}} {{- end -}} {{- if not $imgResource -}} {{- $imgResource = resources.Get "images/consultant.jpg" -}} {{- end -}} {{- $imgURL := "" -}} {{- with $imgResource -}} {{- $imgURL = printf "%s%s" $page.Site.BaseURL (strings.TrimPrefix "/" .RelPermalink) -}} {{- end -}} {{- $canonical := $page.Permalink -}} {{ $title }} {{- /* --- Open Graph (Facebook, LinkedIn, WhatsApp, iMessage) --- */ -}} {{ with $imgURL }} {{ end }} {{ if not $page.IsHome }} {{ with $page.Params.tags }} {{ range . }} {{ end }} {{ end }} {{ end }} {{- /* --- Twitter / X card --- */ -}} {{ with $imgURL }} {{ end }} {{ with $page.Site.Params.twitter }} {{ end }} {{- /* --- Pagination: prev/next rel links for crawlable series --- */ -}} {{- $isListKinds := slice "home" "section" "taxonomy" "term" -}} {{- if in $isListKinds $page.Kind -}} {{- with $page.Paginator -}} {{- if .HasPrev -}} {{- end -}} {{- if .HasNext -}} {{- end -}} {{- end -}} {{- end -}}
